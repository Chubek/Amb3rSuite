cmake_minimum_required(VERSION 3.16)
project(amb3r_sdk_workbook C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_executable(amb3r_sdk_reference examples/host.c examples/plugin.c)
target_include_directories(amb3r_sdk_reference PRIVATE include)
if(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
  target_compile_options(amb3r_sdk_reference PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
enable_testing()
add_test(NAME sdk_reference COMMAND amb3r_sdk_reference)
